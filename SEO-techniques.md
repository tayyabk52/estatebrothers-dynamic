import { NextRequest, NextResponse } from 'next/server';
import { revalidateTag } from 'next/cache';

export async function POST(request: NextRequest) {
  try {
    const payload = await request.json();
    
    // Verify webhook secret from Supabase
    const authHeader = request.headers.get('Authorization');
    if (authHeader !== `Bearer ${process.env.SUPABASE_WEBHOOK_SECRET}`) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Extract property ID from the Supabase update payload
    const propertyId = payload.record.id;
    
    // Invalidate the specific listing and the general listings cache
    revalidateTag(`listing-${propertyId}`, 'max');
    revalidateTag('all-listings', 'max');

    return NextResponse.json({ revalidated: true, now: Date.now() });
  } catch (error) {
    return NextResponse.json({ error: 'Invalid request' }, { status: 400 });
  }
}
Dynamic Routing, Metadata, and Canonicalization StrategyFaceted navigation and deep pagination are critical failure points for large e-commerce and real estate SEO. Without strict controls, search engine crawlers become trapped in near-infinite loops of filtered URLs (e.g., ?beds=3&baths=2&sort=price_asc), wasting finite crawl budget on duplicate or low-value content.To manage pagination effectively, the application must utilize clear, descriptive URL paths or query parameters (e.g., ?page=2) while ensuring each paginated page contains a self-referencing canonical tag pointing exactly to itself. The outdated practice of using the first page of a paginated series as the canonical target for all subsequent pages is a severe anti-pattern; it actively prevents deep-level properties on pages 2, 3, and beyond from being discovered and indexed by Google.For faceted search URLs that generate varying sort orders (e.g., sort by highest price) or highly specific filtered combinations resulting in thin content, the <meta name="robots" content="noindex, follow"> directive must be deployed programmatically. Programmatic XML Sitemap Generation EngineWith a database rapidly expanding to contain tens of thousands of individual properties, societies, phases, and blog posts, relying on static sitemap generation at build time is a critical failure. Furthermore, Google strictly enforces a hard limit of 50,000 URLs per individual XML sitemap file.The Next.js App Router provides a highly optimized solution via the generateSitemaps API. A dedicated sitemap.ts file must be created to programmatically split the massive dataset into manageable chunks.The architecture functions in two stages. First, the generateSitemaps function queries the Supabase database for the total aggregate count of active listings. It divides this integer by 50,000 to determine the required number of sitemap files, returning an array of identifier objects (e.g., [{ id: 0 }, { id: 1 }]).Next, the default sitemap(props) function accepts this dynamic identifier. It calculates the necessary database offset (const start = id * 50000) and the query limit, subsequently querying Supabase for precisely that segmented chunk of URLs. This chunking approach completely prevents memory exhaustion during the generation process, cleanly bypasses Vercel serverless function timeout limits, and provides Google with a perfectly paginated sitemap index.TypeScript// Architectural pattern for chunked sitemap generation in Next.js App Router
import { MetadataRoute } from 'next';

const BASE_URL = process.env.NEXT_PUBLIC_APP_BASE_URL;
const URLS_PER_SITEMAP = 50000;

export async function generateSitemaps() {
  // Query Supabase for total active property count
  const count = await getActivePropertyCount();
  const pages = Math.ceil(count / URLS_PER_SITEMAP);
  
  // Return array of IDs required by Next.js to generate multiple files
  return Array.from({ length: pages }, (_, i) => ({ id: i }));
}

export default async function sitemap({ id }: { id: number }): Promise<MetadataRoute.Sitemap> {
  const start = id * URLS_PER_SITEMAP;
  
  // Fetch only the chunk of properties required for this specific sitemap ID
  const properties = await getPropertiesForSitemap(start, URLS_PER_SITEMAP);
  
  return properties.map((property) => ({
    url: `${BASE_URL}/property/${property.slug}`,
    lastModified: property.updated_at,
    changeFrequency: 'daily',
    priority: 0.8,
  }));
}
